# `docs-search` — POC 1: ADO Wiki / HR-docs style search

**Scope:** the `docs-search` feature only. Assumes the shared `rag-api` and workspace setup
from `000.hackathon-setup.md` are already in place.

**The idea:** flat semantic search over a folder of markdown/text documents (Wiki exports, HR
policy docs, etc.) — no structure, no dependencies, just "find the passage that answers this."

---

## Folder structure

```
docs-search/
├── package.json
├── docs/              # sample .md/.txt files — gitignored, see 000.hackathon-setup.md
└── src/{createIndex.ts, ingest.ts, mcpServer.ts}
```

## `docs-search/src/createIndex.ts`
```ts
import { config } from "dotenv";
config({ path: new URL("../../.env", import.meta.url) });

import { SearchIndexClient, AzureKeyCredential, SearchIndex } from "@azure/search-documents";

const client = new SearchIndexClient(process.env.AZURE_SEARCH_ENDPOINT!, new AzureKeyCredential(process.env.AZURE_SEARCH_KEY!));

const index: SearchIndex = {
  name: "docs-index",
  fields: [
    { name: "id", type: "Edm.String", key: true },
    { name: "content", type: "Edm.String", searchable: true },
    { name: "source", type: "Edm.String", filterable: true },
    { name: "embedding", type: "Collection(Edm.Single)", searchable: true,
      vectorSearchDimensions: 384, vectorSearchProfileName: "default-profile" },
  ],
  vectorSearch: {
    algorithms: [{ name: "default-hnsw", kind: "hnsw" }],
    profiles: [{ name: "default-profile", algorithmConfigurationName: "default-hnsw" }],
  },
};

await client.createOrUpdateIndex(index);
console.log("docs-index ready.");
```

## `docs-search/src/ingest.ts`

Walks `docs/`, chunks every `.md`/`.txt` file, embeds locally, uploads directly to
`docs-index` (write path — straight to `@azure/search-documents`, never through `rag-api`).
Chunking here is paragraph-aware, unlike `doc-hierarchy`'s fixed-size split — a better fit for
longer free-form markdown than for short structured documents.

```ts
import { config } from "dotenv";
config({ path: new URL("../../.env", import.meta.url) });

import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { pipeline, type FeatureExtractionPipeline } from "@huggingface/transformers";
import { SearchClient, AzureKeyCredential } from "@azure/search-documents";

const packageRoot = fileURLToPath(new URL("..", import.meta.url));
const docsDir = path.join(packageRoot, "docs");

type Doc = { id: string; content: string; source: string; embedding: number[] };

const client = new SearchClient<Doc>(
  process.env.AZURE_SEARCH_ENDPOINT!, "docs-index",
  new AzureKeyCredential(process.env.AZURE_SEARCH_KEY!)
);

let extractorPromise: Promise<FeatureExtractionPipeline> | undefined;
async function embed(text: string): Promise<number[]> {
  extractorPromise ??= pipeline("feature-extraction", "Xenova/all-MiniLM-L6-v2");
  const extractor = await extractorPromise;
  return Array.from((await extractor(text, { pooling: "mean", normalize: true })).data as Float32Array);
}

function odataEscape(s: string): string {
  return s.replace(/'/g, "''");
}

// Groups whole paragraphs up to ~maxSize instead of cutting mid-sentence.
function chunk(text: string, maxSize = 1000): string[] {
  const paragraphs = text.split(/\n\s*\n/).filter(p => p.trim());
  const out: string[] = [];
  let current = "";
  for (const p of paragraphs) {
    if (current && current.length + p.length + 2 > maxSize) {
      out.push(current);
      current = p;
    } else {
      current = current ? `${current}\n\n${p}` : p;
    }
  }
  if (current) out.push(current);
  return out;
}

function walk(dir: string): string[] {
  return fs.readdirSync(dir, { withFileTypes: true }).flatMap(e => {
    const full = path.join(dir, e.name);
    if (e.isDirectory()) return walk(full);
    return /\.(md|txt)$/.test(e.name) ? [full] : [];
  });
}

// Azure Search document keys only allow letters, digits, dash, underscore, and equals —
// a file path's slashes and dots need sanitizing before use as an ID.
function safeId(source: string, index: number): string {
  return `${source.replace(/[^a-zA-Z0-9\-_=]/g, "_")}-${index}`;
}

async function ingestFile(filePath: string) {
  const source = path.relative(packageRoot, filePath).split(path.sep).join("/");
  const text = fs.readFileSync(filePath, "utf-8");

  const stale = await client.search("*", { filter: `source eq '${odataEscape(source)}'`, select: ["id"] });
  const staleIds: string[] = [];
  for await (const r of stale.results) staleIds.push(r.document.id);
  if (staleIds.length) await client.deleteDocuments("id", staleIds);

  const chunks = chunk(text);
  const documents = await Promise.all(chunks.map(async (c, i) => ({
    id: safeId(source, i),
    content: c,
    source,
    embedding: await embed(c),
  })));
  await client.uploadDocuments(documents);
  console.log(`Indexed ${documents.length} chunks from ${source}`);
}

for (const file of walk(docsDir)) {
  await ingestFile(file);
}
```

## `docs-search/src/mcpServer.ts`
```ts
import { McpServer } from "@modelcontextprotocol/sdk/server/mcp.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import { z } from "zod";

const RAG_API_URL = process.env.RAG_API_URL ?? "http://localhost:8000/search";
const server = new McpServer({ name: "docs-search", version: "1.0.0" });

server.registerTool(
  "search_docs",
  {
    description: "Search the local knowledge base for relevant passages.",
    inputSchema: { query: z.string() },
  },
  async ({ query }) => {
    const res = await fetch(RAG_API_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ index: "docs-index", query, top_k: 5 }),
    });
    const { results } = (await res.json()) as { results: { source: string; content: string }[] };
    const text = results.length
      ? results.map(r => `Source: ${r.source}\n${r.content}`).join("\n\n---\n\n")
      : "No relevant documents found.";
    return { content: [{ type: "text" as const, text }] };
  }
);

const transport = new StdioServerTransport();
await server.connect(transport);
```

## Install and run

**`docs-search/package.json`**:
```json
{
  "name": "docs-search",
  "private": true,
  "type": "module",
  "scripts": {
    "create-index": "tsx src/createIndex.ts",
    "ingest": "tsx src/ingest.ts",
    "mcp": "tsx src/mcpServer.ts"
  }
}
```

```bash
npm install @modelcontextprotocol/sdk zod @azure/search-documents @huggingface/transformers dotenv -w docs-search

npm run docs-search:create-index   # from the repo root
npm run docs-search:ingest
```
